;-------------------------------------------------------------------------
; PADRÃO PARA MANIPULAÇÃO DE DADOS (STEP-BY-STEP)
;-------------------------------------------------------------------------
; TIPO: Define a largura (byte, word, dword)
; TAMANHO: O valor em bytes do tipo (1, 2, 4)
; ÍNDICE: A posição lógica do elemento (0, 1, 2... n)
;
; DESLOCAMENTO VARIÁVEL: índice * tamanho
; DESLOCAMENTO FIXO (disp): constante somada ao endereço
; DESLOCAMENTO TOTAL: (índice * tamanho) + disp
;
; REGRA GERAL DE ENDEREÇAMENTO:
;    [base + (índice * tamanho) + disp]
;
; CASO MAIS SIMPLES (sem disp):
;    [base + índice * tamanho]
;
; 1. Preenchimento:
;    mov tipo [buffer + 0*tamanho + 0], dado_0
;    ...
;    mov tipo [buffer + n*tamanho + 0], dado_n
;
; 2. Preenchimento com deslocamento fixo:
;    mov tipo [buffer + índice*tamanho + disp], dado
;    ; Ex: pular cabeçalho, alinhar, etc.
;
; 3. Saída (Kernel):
;    mov ecx, buffer                         ; endereço base
;    mov edx, (n + 1) * tamanho              ; total de bytes

; ENDEREÇO = BASE + DESLOCAMENTO(S); TODO acesso à memória é uma soma de deslocamentos
